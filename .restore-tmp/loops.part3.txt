async function trackGenerationMilestone({
  email,
  userId,
  pluginId,
  pluginVersion,
  generationsCount,
  imagesUnprocessed
}) {
  if (generationsCount <= 0 || generationsCount % 5 !== 0) return;
  const timestamp = new Date().toISOString();
  await upsertPluginContact({
    email,
    userId,
    pluginId,
    pluginVersion,
    timestamp,
    extra: { generationsCount }
  });
  await sendEvent('generation_completed', {
    email,
    userId,
    pluginId,
    pluginVersion,
    idempotencyParts: [generationsCount],
    generationsCount,
    imagesUnprocessed: imagesUnprocessed || 0,
    lastGenerationAt: timestamp
  });
}

// Called only after the caller has confirmed the site's wallet is empty and
// the site has no active, trialing or past_due subscription (see
// fresh-stack/services/usage.js resolvePaywallEligibility). The plan is the
// account's real plan id; it is never assumed to be 'free'.
async function trackCreditsExhausted({
  email,
  userId,
  pluginId,
  pluginVersion,
  imagesUnprocessed,
  periodStart,
  plan,
  paywallHitAt = new Date().toISOString()
}) {
  if (isLoopsSkippedEmail(email)) {
    logSkippedEmail(email, { path: '/events/send', method: 'POST', event_name: 'credits_exhausted' });
    return null;
  }
  const plugin = getPlugin(pluginId);
  const planId = typeof plan === 'string' && plan.trim() ? plan.trim().toLowerCase() : 'unknown';
  // Loops workflows branch on contact properties, so record which plugin hit
  // the paywall on the contact before the event that starts the workflow.
  await loopsRequest('PUT', '/contacts/update', {
    email,
    ...(userId ? { userId: String(userId) } : {}),
    paywallPluginId: plugin.id,
    paywallHitAt,
    plan: planId
  });
  return sendEvent('credits_exhausted', {
    email,
    userId,
    pluginId: plugin.id,
    pluginVersion,
    // Once per account, plugin and billing period (Loops idempotency key).
    idempotencyParts: [periodStart || 'current'],
    imagesUnprocessed: imagesUnprocessed || 0,
    plan: planId,
    paywallHitAt
  });
}

async function trackPlanUpgraded({
  email,
  userId,
  planName,
  pluginId = 'alt_text',
  pluginVersion,
  purchaseType = 'new_purchase',
  billingPeriod = 'unknown',
  amount = null,
  currency = null,
  stripeEventId = null
}) {
  await loopsRequest('PUT', '/contacts/update', {
    email,
    ...(userId ? { userId: String(userId) } : {}),
    plan: planName,
    // Subscription updates can report a downgrade to free; only mark real paid plans.
    ...(planName && String(planName).toLowerCase() !== 'free' ? { userGroup: 'paid' } : {})
  });
  await sendEvent('plan_upgraded', {
    email,
    userId,
    pluginId,
    pluginVersion,
    idempotencyParts: [stripeEventId || planName],
    plan: planName,
    purchaseType,
    billingPeriod,
    amount,
    currency,
    stripeEventId
  });
}

async function trackPaymentFailed({
  email,
  planName = null,
  amount = null,
  currency = null,
  failureCode = null,
  declineCode = null,
  recoverability = 'recoverable',
  paymentIntentId = null,
  chargeId = null,
  paymentLinkId = null,
  checkoutSessionId = null,
  stripeEventId = null
}) {
  const failedAt = new Date().toISOString();
  await loopsRequest('PUT', '/contacts/update', {
    email,
    lastPaymentFailureAt: failedAt,
    lastPaymentFailurePlan: planName || '',
    lastPaymentFailureCode: failureCode || declineCode || '',
    lastPaymentFailureRecoverability: recoverability
  });
  await loopsRequest('POST', '/events/send', {
    email,
    eventName: 'payment_failed',
    eventProperties: {
      plan: planName,
      amount,
      currency,
      failureCode,
      declineCode,
      recoverability,
      lastPaymentFailureRecoverability: recoverability,
      paymentIntentId,
      chargeId,
      paymentLinkId,
      checkoutSessionId,
      stripeEventId
    }
  }, { idempotencyKey: stripeEventId });
}
